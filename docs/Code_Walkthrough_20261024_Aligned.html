"""预算对齐的新实验：固定旧样本 → 本地压缩 → API 改写 → 全部重新预测 → 审计。

各阶段可续跑；旧目录只读。示例：

"""
import argparse
import gc
import math
import os
import socket
import sys
import time
import threading
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timezone
from pathlib import Path
from unittest.mock import patch

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))
from blackwell.budgeting import align_local, select_candidate
from blackwell.client import CachedChatClient, parse_probability
from blackwell.compressors import LOCAL_METHODS, create_compressor
from blackwell.compressors.llm_rewrite import API_METHODS, LLMRewriteCompressor, rewrite_messages
from blackwell.diagnostics import probability_diagnostics, retention_diagnostics
from blackwell.prompts import prediction_messages
from blackwell.storage import file_hash, fingerprint, read_json, write_json
from run_experiment import code_provenance, load_config, load_local_keys, write_outputs

METHODS = ['full', 'no_context', 'head_tail', *LOCAL_METHODS, *API_METHODS]


def compression_config(config):
    """只覆盖生成器的模型、超时和尝试上限；不修改 reader 配置。"""
    return {**config, 'model': config['compression_model'],
            'max_api_attempts': config['compression_max_api_attempts'],
            'timeout_seconds': config.get('compression_timeout_seconds', 600)}


def require(condition, message):
    if not condition:
        raise ValueError(message)


def prepare(config):
    """固定既有样本和源文件哈希；重新预测全部条件，避免混用旧日期的 forecasts。"""
    from blackwell.data import load_dataset
    from blackwell.compressors.runtime import configure_cache, runtime_identity, verify_assets
    from blackwell.tokenization import ContextTokenizer
    configure_cache()
    verify_assets(LOCAL_METHODS)
    require(config['methods'] == METHODS, 'Unexpected paired methods.')
    require(config['retained_ratio'] == .5, 'This protocol fixes 50% token cap.')
    source = ROOT / 'runs' / config['source_run']
    require(source.parent == ROOT / 'runs' and source.name != config['run_name'], 'Unsafe source path.')
    old = read_json(source / 'manifest.json')
    require(old['status'] == 'complete', 'Source run incomplete.')
    examples = read_json(source / 'examples.json')
    reloaded, provenance = load_dataset(old['identity']['config'], offline=True)
    require(examples == reloaded and fingerprint(examples) == old['identity']['examples_sha256'], 'Source samples changed.')
    require(config['sample_size'] <= len(examples), 'Cannot expand the frozen source population.')
    examples = examples[:config['sample_size']]
    snapshots = {p.name: file_hash(p) for p in source.iterdir() if p.is_file()}
    code = code_provenance()
    for path in [Path(__file__), *sorted((ROOT / 'blackwell/compressors/prompts').glob('*'))]:
        code['files_sha256'][path.relative_to(ROOT).as_posix()] = file_hash(path)
    runtime = runtime_identity(LOCAL_METHODS, config['compression_device'])
    runtime['budget_policy'] = 'length-only native-rate search <=8 calls; prefer 95-100% cap band, else closest length (feasible wins ties); explicitly head-tail clamp selected overshoots'
    identity = {'config': config, 'dataset': provenance, 'examples_sha256': fingerprint(examples),
                'code': code['files_sha256'], 'source_files_sha256': snapshots,
                'local_compressors': runtime}
    run = ROOT / 'runs' / config['run_name']
    if (run / 'manifest.json').exists():
        manifest = read_json(run / 'manifest.json')
        require(manifest['identity'] == identity, 'Code/config/data changed: choose a new run_name.')
        require(read_json(run / 'examples.json') == examples, 'Saved examples changed.')
    else:
        manifest = {'identity': identity, 'created_utc': datetime.now(timezone.utc).isoformat(), 'status': 'prepared'}
        write_json(run / 'manifest.json', manifest)
        write_json(run / 'examples.json', examples)
    return run, ContextTokenizer(), examples, manifest


def compression_request(context, budget, method, manifest):
    """压缩缓存键只含上下文、预算和固定协议；绝不传整个 example。"""
    identity = manifest['identity']
    return {'context': context, 'budget': budget, 'method': method,
            'protocol_sha256': fingerprint({'code': identity['code'],
                                           'runtime': identity['local_compressors'],
                                           'model': identity['config']['compression_model']})}


def read_compaction(run, request):
    key = fingerprint(request)
    path = run / 'aligned_compression_cache' / f'{key}.json'
    if not path.exists():
        return None
    entry = read_json(path)
    require(entry['request'] == request and entry['sha256'] == fingerprint(entry['result']), 'Compression cache corruption.')
    return {**entry['result'], 'local_compression_hash': key}


def save_compaction(run, request, result, seconds):
    result = {**result, 'compression_seconds': seconds}
    write_json(run / 'aligned_compression_cache' / f'{fingerprint(request)}.json',
               {'request': request, 'result': result, 'sha256': fingerprint(result)})


def compress_stage(config, run, tokenizer, examples, manifest, api=False):
    """先按方法遍历，GPU 一次只装一个模型；相同原文只压缩一次。"""
    contexts = list(dict.fromkeys(e['context'] for e in examples))
    if api:
        workers = min(config.get('compression_workers', 1), len(contexts))
        require(type(workers) is int and 1 <= workers <= 8, 'Use 1–8 rewrite workers.')
        stop = threading.Event()
        with ThreadPoolExecutor(max_workers=workers) as pool:
            jobs = [pool.submit(rewrite_worker, config, run, tokenizer, contexts[w::workers], manifest, w, workers, stop)
                    for w in range(workers)]
            for job in jobs:
                job.result()
        return
    for method in LOCAL_METHODS:
        engine = None
        for index, context in enumerate(contexts, 1):
            budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
            request = compression_request(context, budget, method, manifest)
            if read_compaction(run, request) is not None:
                continue
            if engine is None:
                # LLMLingua 的 seed_everything 会改环境字符串，但不会改变已启动进程的 hash seed。
                # 先核实进程从 seed=0 启动，再恢复字符串，供 SALT 的防误用检查读取。
                require(not sys.flags.hash_randomization, 'Start with PYTHONHASHSEED=0.')
                os.environ['PYTHONHASHSEED'] = '0'
                engine = create_compressor(method, config['compression_device'])
            start = time.perf_counter()
            result = align_local(engine, context, budget, tokenizer)
            save_compaction(run, request, result, time.perf_counter()-start)
            print(f'{method} [{index}/{len(contexts)}] {result["tokens"]}/{budget} tokens; clip={result["post_clip_used"]}', flush=True)
        del engine
        gc.collect()
        import torch
        torch.cuda.empty_cache()


def rewrite_worker(config, run, tokenizer, contexts, manifest, worker, workers, stop):
    """每个 worker 串行处理自己的原文分片；没有共享可写 API 账本。"""
    from blackwell.rewrite_client import PacedRewriteClient
    relative = f'compression_api/worker_{worker}'
    client = PacedRewriteClient(compression_config(config), run/relative, tokenizer, workers, worker)
    try:
        for method in API_METHODS:
            engine = LLMRewriteCompressor(client, tokenizer, method)
            for index, context in enumerate(contexts, 1):
                if stop.is_set():
                    return
                budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
                request = compression_request(context, budget, method, manifest)
                if read_compaction(run, request) is not None:
                    continue
                start = time.perf_counter()
                result = engine.compress(context, budget)
                result['compression_api_subdir'] = relative
                save_compaction(run, request, result, time.perf_counter()-start)
                print(f'{method} worker {worker} [{index}/{len(contexts)}]: {result["tokens"]}/{budget}, clip={result["post_clip_used"]}', flush=True)
    except Exception:
        stop.set()
        raise


def condition_text(context, method, tokenizer, run, manifest):
    """预测阶段只读已完成的压缩结果，不再加载压缩模型或修改压缩文本。"""
    original = tokenizer.count(context)
    budget = max(1, math.floor(.5*original))
    if method in LOCAL_METHODS or method in API_METHODS:
        result = read_compaction(run, compression_request(context, budget, method, manifest))
        require(result is not None, f'Missing compression: {method}')
        # 候选全文保存在压缩缓存，不在每道问题中重复复制。
        return {k: v for k, v in result.items() if k != 'candidates'}
    text = context if method == 'full' else ''
    if method == 'head_tail':
        text = tokenizer.head_tail(context, budget)
        return select_candidate([{'text': text, 'tokens': tokenizer.count(text)}], budget, tokenizer)
    return {'text': text, 'tokens': tokenizer.count(text), 'budget_tokens': None,
            'budget_met': None, 'length_repair_used': False, 'compression_request_hashes': []}


def predict(config, run, tokenizer, examples, manifest, offline=False):
    client = CachedChatClient(config, run, offline=offline)
    rows = []
    # 全部压缩完成后才访问 question / label；每题的所有条件配对。
    for index, example in enumerate(examples, 1):
        context = example['context']
        original = tokenizer.count(context)
        conditions = {}
        for method in config['methods']:
            c = condition_text(context, method, tokenizer, run, manifest)
            raw, key = client.complete(prediction_messages(c['text'], example['question']), config['prediction_max_tokens'], json_mode=True)
            probability = parse_probability(raw)
            conditions[method] = {**c, 'p_yes': probability, 'predicted_label': int(probability >= .5),
                                  'prediction_request_hash': key, 'retained_ratio_actual': c['tokens']/original}
        rows.append({**example, 'original_tokens': original, 'conditions': conditions})
        if not offline:
            write_json(run / 'predictions.json', rows)
            print(f'Predict [{index}/{len(examples)}] '+example['id'], flush=True)
    return rows


def finish(config, run, rows, manifest):
    metrics = write_outputs(run, rows, config)
    labels = [row['label'] for row in rows]
    diagnostics = {}
    for method in config['methods']:
        conditions = [row['conditions'][method] for row in rows]
        diagnostics[method] = {'probability': probability_diagnostics([c['p_yes'] for c in conditions], labels),
                               'retention': retention_diagnostics(conditions)}
    write_json(run / 'diagnostics.json', diagnostics)
    manifest.update(status='complete', completed_utc=datetime.now(timezone.utc).isoformat())
    write_json(run / 'manifest.json', manifest)
    return metrics


def audit(config, run, tokenizer, examples, manifest):
    """重新构造每条 API 请求，核验长度与 query 边界，并独立验证全部 InfoGap 单元格。"""
    from tools.extend_with_salt import check_cached_prediction, check_matrix
    rows = read_json(run / 'predictions.json')
    require(len(rows) == len(examples), 'Incomplete sample set.')
    source = ROOT / 'runs' / config['source_run']
    for name, digest in manifest['identity']['source_files_sha256'].items():
        require(file_hash(source / name) == digest, 'Old source file changed: '+name)
    checked = set()
    for row, example in zip(rows, examples):
        require({k: row[k] for k in example} == example, 'Sample pairing changed.')
        require(list(row['conditions']) == config['methods'], 'Condition mismatch.')
        for method, condition in row['conditions'].items():
            require(tokenizer.count(condition['text']) == condition['tokens'], 'Token count mismatch.')
            check_cached_prediction(run, config, row, condition)
            expected = condition_text(row['context'], method, tokenizer, run, manifest)
            require(all(condition[k] == v for k, v in expected.items()), 'Condition differs from compression cache.')
            if method in {'full', 'no_context', 'head_tail'}:
                continue
            request = compression_request(row['context'], condition['budget_tokens'], method, manifest)
            cached = read_compaction(run, request)
            key = fingerprint(request)
            if key in checked:
                continue
            checked.add(key)
            candidates = cached['candidates']
            require(all(tokenizer.count(c['text']) == c['tokens'] for c in candidates), 'Candidate length mismatch.')
            selection = select_candidate(candidates, condition['budget_tokens'], tokenizer)
            require(all(cached[k] == v for k, v in selection.items() if k != 'compression_request_hashes'), 'Length-only candidate selection mismatch.')
            if method in API_METHODS:
                previous, previous_tokens = None, None
                for candidate in candidates:
                    messages = rewrite_messages(row['context'], condition['budget_tokens'], method, previous, previous_tokens)
                    raw, request_hash = CachedChatClient(compression_config(config), run / cached['compression_api_subdir'], offline=True).complete(
                        messages, max_tokens=min(32768, max(512, 2*condition['budget_tokens']+512)))
                    require(raw == candidate['text'] and request_hash == candidate['request_hash'], 'Rewrite request/response mismatch.')
                    previous, previous_tokens = raw, candidate['tokens']
    check_matrix(rows, read_json(run / 'metrics.json'), config['methods'])
    before = {name: file_hash(run / name) for name in ('predictions.json', 'metrics.json', 'api_attempts.json')}
    with patch.object(socket.socket, 'connect', side_effect=AssertionError('Offline replay')):
        replayed = predict(config, run, tokenizer, examples, manifest, offline=True)
    require(replayed == rows, 'Offline predictions changed.')
    require(before == {name: file_hash(run / name) for name in before}, 'Replay changed files.')
    result = {'status': 'passed', 'samples': len(rows), 'forecasts': len(rows)*len(METHODS),
              'compression_contexts_checked': len(checked), 'infogap_cells': len(METHODS)**2,
              'old_files_unchanged': True, 'offline_replay_identical': True,
              'scope': 'Full request reconstruction; compression inputs contain context/length only; original forecasts unmodified.'}
    write_json(run / 'audit.json', result)
    print(result, flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--config', required=True)
    parser.add_argument('--stage', choices=['prepare', 'local', 'rewrite', 'predict', 'audit', 'all'], default='all')
    args = parser.parse_args()
    config = load_config(ROOT / args.config)
    with patch.object(socket.socket, 'connect', side_effect=AssertionError('Preparation must be offline')):
        run, tokenizer, examples, manifest = prepare(config)
    load_local_keys()
    try:
        if args.stage in {'all', 'local'}:
            manifest['status'] = 'local_compression'
            write_json(run / 'manifest.json', manifest)
            with patch.object(socket.socket, 'connect', side_effect=AssertionError('Local compression must be offline')):
                compress_stage(config, run, tokenizer, examples, manifest)
        if args.stage in {'all', 'rewrite'}:
            manifest['status'] = 'api_compression'
            write_json(run / 'manifest.json', manifest)
            compress_stage(config, run, tokenizer, examples, manifest, api=True)
        if args.stage in {'all', 'predict'}:
            manifest['status'] = 'predicting'
            write_json(run / 'manifest.json', manifest)
            rows = predict(config, run, tokenizer, examples, manifest)
            finish(config, run, rows, manifest)
        if args.stage in {'all', 'audit'}:
            audit(config, run, tokenizer, examples, manifest)
    except Exception as error:
        manifest.update(status='failed', error_type=type(error).__name__)
        write_json(run / 'manifest.json', manifest)
        raise


if __name__ == '__main__':
    main()

import math


def select_candidate(candidates, budget, tokenizer):
    """优先选择目标区间内最长候选；否则选距离预算最近者，必要时显式裁剪。"""
    in_band = [c for c in candidates if math.ceil(.95*budget) <= c['tokens'] <= budget]
    selected = (max(in_band, key=lambda c: c['tokens']) if in_band else
                min(candidates, key=lambda c: (abs(c['tokens']-budget), c['tokens'] > budget)))
    text = selected['text']
    clipped = tokenizer.count(text) > budget
    if clipped:
        text = tokenizer.head_tail(text, budget)
    tokens = tokenizer.count(text)
    return {'text': text, 'tokens': tokens, 'budget_tokens': budget,
            'budget_met': tokens <= budget, 'budget_fill': tokens / budget,
            'within_target_band': math.ceil(.95 * budget) <= tokens <= budget,
            'post_clip_used': clipped, 'selected_candidate': candidates.index(selected),
            'length_repair_used': len(candidates) > 1, 'compression_request_hashes': []}


def align_local(engine, context, budget, tokenizer, max_trials=8):
    """用最多 max_trials 次原生参数试探逼近预算；不假定输出长度严格单调。"""
    original = tokenizer.count(context)
    low, high = .001, .999999
    rate = min(high, max(low, budget / original))
    candidates = []
    for _ in range(max_trials):
        text = engine.compress(context, retained_ratio=rate)
        candidate = {'text': text, 'tokens': tokenizer.count(text), 'native_rate': rate}
        candidates.append(candidate)
        if math.ceil(.95 * budget) <= candidate['tokens'] <= budget:
            break
        if candidate['tokens'] > budget:
            high = rate
        else:
            low = rate
        rate = (low + high) / 2
    result = select_candidate(candidates, budget, tokenizer)
    return {**result, 'candidates': candidates,
            'native_retained_ratio': candidates[result['selected_candidate']]['native_rate']}

    """只接收原文、长度、方法和此前草稿；新增长度约束明确覆盖原生 TE 的自适应目标。"""

        path = PROMPT_DIR / 'telegrapher_v5.txt'

    else:
        raise ValueError(f'Unknown API compressor: {method}')

    runtime = runtime_identity(LOCAL_METHODS, config['compression_device'])
    runtime['budget_policy'] = 'length-only native-rate search <=8 calls; prefer 95-100% cap band, else closest length (feasible wins ties); explicitly head-tail clamp selected overshoots'
    identity = {'config': config, 'dataset': provenance, 'examples_sha256': fingerprint(examples),
                'code': code['files_sha256'], 'source_files_sha256': snapshots,
                'local_compressors': runtime}
    run = ROOT / 'runs' / config['run_name']
    if (run / 'manifest.json').exists():
        manifest = read_json(run / 'manifest.json')
        require(manifest['identity'] == identity, 'Code/config/data changed: choose a new run_name.')
        require(read_json(run / 'examples.json') == examples, 'Saved examples changed.')
    else:
        manifest = {'identity': identity, 'created_utc': datetime.now(timezone.utc).isoformat(), 'status': 'prepared'}
        write_json(run / 'manifest.json', manifest)
        write_json(run / 'examples.json', examples)

def compression_request(context, budget, method, manifest):
    """压缩缓存键只含上下文、预算和固定协议；绝不传整个 example。"""
    identity = manifest['identity']
    return {'context': context, 'budget': budget, 'method': method,
            'protocol_sha256': fingerprint({'code': identity['code'],
                                           'runtime': identity['local_compressors'],
                                           'model': identity['config']['compression_model']})}


def read_compaction(run, request):
    key = fingerprint(request)
    path = run / 'aligned_compression_cache' / f'{key}.json'
    if not path.exists():
        return None
    entry = read_json(path)
    require(entry['request'] == request and entry['sha256'] == fingerprint(entry['result']), 'Compression cache corruption.')
    return {**entry['result'], 'local_compression_hash': key}


def save_compaction(run, request, result, seconds):
    result = {**result, 'compression_seconds': seconds}
    write_json(run / 'aligned_compression_cache' / f'{fingerprint(request)}.json',
               {'request': request, 'result': result, 'sha256': fingerprint(result)})


def compress_stage(config, run, tokenizer, examples, manifest, api=False):
    """先按方法遍历，GPU 一次只装一个模型；相同原文只压缩一次。"""
    contexts = list(dict.fromkeys(e['context'] for e in examples))
    if api:
        workers = min(config.get('compression_workers', 1), len(contexts))
        require(type(workers) is int and 1 <= workers <= 8, 'Use 1–8 rewrite workers.')
        stop = threading.Event()
        with ThreadPoolExecutor(max_workers=workers) as pool:
            jobs = [pool.submit(rewrite_worker, config, run, tokenizer, contexts[w::workers], manifest, w, workers, stop)
                    for w in range(workers)]
            for job in jobs:
                job.result()
        return
    for method in LOCAL_METHODS:
        engine = None
        for index, context in enumerate(contexts, 1):
            budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
            request = compression_request(context, budget, method, manifest)
            if read_compaction(run, request) is not None:
                continue
            if engine is None:
                # LLMLingua 的 seed_everything 会改环境字符串，但不会改变已启动进程的 hash seed。
                # 先核实进程从 seed=0 启动，再恢复字符串，供 SALT 的防误用检查读取。
                require(not sys.flags.hash_randomization, 'Start with PYTHONHASHSEED=0.')
                os.environ['PYTHONHASHSEED'] = '0'
                engine = create_compressor(method, config['compression_device'])
            start = time.perf_counter()
            result = align_local(engine, context, budget, tokenizer)
            save_compaction(run, request, result, time.perf_counter()-start)
            print(f'{method} [{index}/{len(contexts)}] {result["tokens"]}/{budget} tokens; clip={result["post_clip_used"]}', flush=True)
        del engine
        gc.collect()
        import torch
        torch.cuda.empty_cache()


def rewrite_worker(config, run, tokenizer, contexts, manifest, worker, workers, stop):
    """每个 worker 串行处理自己的原文分片；没有共享可写 API 账本。"""
    from blackwell.rewrite_client import PacedRewriteClient
    relative = f'compression_api/worker_{worker}'
    client = PacedRewriteClient(compression_config(config), run/relative, tokenizer, workers, worker)
    try:
        for method in API_METHODS:
            engine = LLMRewriteCompressor(client, tokenizer, method)
            for index, context in enumerate(contexts, 1):
                if stop.is_set():
                    return
                budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
                request = compression_request(context, budget, method, manifest)
                if read_compaction(run, request) is not None:
                    continue
                start = time.perf_counter()
                result = engine.compress(context, budget)
                result['compression_api_subdir'] = relative
                save_compaction(run, request, result, time.perf_counter()-start)
                print(f'{method} worker {worker} [{index}/{len(contexts)}]: {result["tokens"]}/{budget}, clip={result["post_clip_used"]}', flush=True)
    except Exception:
        stop.set()
        raise


def condition_text(context, method, tokenizer, run, manifest):
    """预测阶段只读已完成的压缩结果，不再加载压缩模型或修改压缩文本。"""
    original = tokenizer.count(context)
    budget = max(1, math.floor(.5*original))
    if method in LOCAL_METHODS or method in API_METHODS:
        result = read_compaction(run, compression_request(context, budget, method, manifest))
        require(result is not None, f'Missing compression: {method}')
        # 候选全文保存在压缩缓存，不在每道问题中重复复制。
        return {k: v for k, v in result.items() if k != 'candidates'}
    text = context if method == 'full' else ''
    if method == 'head_tail':
        text = tokenizer.head_tail(context, budget)
        return select_candidate([{'text': text, 'tokens': tokenizer.count(text)}], budget, tokenizer)
    return {'text': text, 'tokens': tokenizer.count(text), 'budget_tokens': None,
            'budget_met': None, 'length_repair_used': False, 'compression_request_hashes': []}


def predict(config, run, tokenizer, examples, manifest, offline=False):
    client = CachedChatClient(config, run, offline=offline)
    rows = []
    # 全部压缩完成后才访问 question / label；每题的所有条件配对。
    for index, example in enumerate(examples, 1):
        context = example['context']
        original = tokenizer.count(context)
        conditions = {}
        for method in config['methods']:
            c = condition_text(context, method, tokenizer, run, manifest)
            raw, key = client.complete(prediction_messages(c['text'], example['question']), config['prediction_max_tokens'], json_mode=True)
            probability = parse_probability(raw)
            conditions[method] = {**c, 'p_yes': probability, 'predicted_label': int(probability >= .5),
                                  'prediction_request_hash': key, 'retained_ratio_actual': c['tokens']/original}
        rows.append({**example, 'original_tokens': original, 'conditions': conditions})
        if not offline:
            write_json(run / 'predictions.json', rows)
            print(f'Predict [{index}/{len(examples)}] '+example['id'], flush=True)
    return rows


def finish(config, run, rows, manifest):
    metrics = write_outputs(run, rows, config)
    labels = [row['label'] for row in rows]
    diagnostics = {}
    for method in config['methods']:
        conditions = [row['conditions'][method] for row in rows]
        diagnostics[method] = {'probability': probability_diagnostics([c['p_yes'] for c in conditions], labels),
                               'retention': retention_diagnostics(conditions)}
    write_json(run / 'diagnostics.json', diagnostics)
    manifest.update(status='complete', completed_utc=datetime.now(timezone.utc).isoformat())
    write_json(run / 'manifest.json', manifest)
    return metrics


def audit(config, run, tokenizer, examples, manifest):
    """重新构造每条 API 请求，核验长度与 query 边界，并独立验证全部 InfoGap 单元格。"""
    from tools.extend_with_salt import check_cached_prediction, check_matrix
    rows = read_json(run / 'predictions.json')
    require(len(rows) == len(examples), 'Incomplete sample set.')
    source = ROOT / 'runs' / config['source_run']
    for name, digest in manifest['identity']['source_files_sha256'].items():
        require(file_hash(source / name) == digest, 'Old source file changed: '+name)
    checked = set()
    for row, example in zip(rows, examples):
        require({k: row[k] for k in example} == example, 'Sample pairing changed.')
        require(list(row['conditions']) == config['methods'], 'Condition mismatch.')
        for method, condition in row['conditions'].items():
            require(tokenizer.count(condition['text']) == condition['tokens'], 'Token count mismatch.')
            check_cached_prediction(run, config, row, condition)
            expected = condition_text(row['context'], method, tokenizer, run, manifest)
            require(all(condition[k] == v for k, v in expected.items()), 'Condition differs from compression cache.')
            if method in {'full', 'no_context', 'head_tail'}:
                continue
            request = compression_request(row['context'], condition['budget_tokens'], method, manifest)
            cached = read_compaction(run, request)
            key = fingerprint(request)
            if key in checked:
                continue
            checked.add(key)
            candidates = cached['candidates']
            require(all(tokenizer.count(c['text']) == c['tokens'] for c in candidates), 'Candidate length mismatch.')
            selection = select_candidate(candidates, condition['budget_tokens'], tokenizer)
            require(all(cached[k] == v for k, v in selection.items() if k != 'compression_request_hashes'), 'Length-only candidate selection mismatch.')
            if method in API_METHODS:
                previous, previous_tokens = None, None
                for candidate in candidates:
                    messages = rewrite_messages(row['context'], condition['budget_tokens'], method, previous, previous_tokens)
                    raw, request_hash = CachedChatClient(compression_config(config), run / cached['compression_api_subdir'], offline=True).complete(
                        messages, max_tokens=min(32768, max(512, 2*condition['budget_tokens']+512)))
                    require(raw == candidate['text'] and request_hash == candidate['request_hash'], 'Rewrite request/response mismatch.')
                    previous, previous_tokens = raw, candidate['tokens']
    check_matrix(rows, read_json(run / 'metrics.json'), config['methods'])
    before = {name: file_hash(run / name) for name in ('predictions.json', 'metrics.json', 'api_attempts.json')}
    with patch.object(socket.socket, 'connect', side_effect=AssertionError('Offline replay')):
        replayed = predict(config, run, tokenizer, examples, manifest, offline=True)
    require(replayed == rows, 'Offline predictions changed.')
    require(before == {name: file_hash(run / name) for name in before}, 'Replay changed files.')
    result = {'status': 'passed', 'samples': len(rows), 'forecasts': len(rows)*len(METHODS),
              'compression_contexts_checked': len(checked), 'infogap_cells': len(METHODS)**2,
              'old_files_unchanged': True, 'offline_replay_identical': True,
              'scope': 'Full request reconstruction; compression inputs contain context/length only; original forecasts unmodified.'}
    write_json(run / 'audit.json', result)
    print(result, flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--config', required=True)
    parser.add_argument('--stage', choices=['prepare', 'local', 'rewrite', 'predict', 'audit', 'all'], default='all')
    args = parser.parse_args()
    config = load_config(ROOT / args.config)
    with patch.object(socket.socket, 'connect', side_effect=AssertionError('Preparation must be offline')):
        run, tokenizer, examples, manifest = prepare(config)
    load_local_keys()
    try:
        if args.stage in {'all', 'local'}:
            manifest['status'] = 'local_compression'
            write_json(run / 'manifest.json', manifest)
            with patch.object(socket.socket, 'connect', side_effect=AssertionError('Local compression must be offline')):
                compress_stage(config, run, tokenizer, examples, manifest)
        if args.stage in {'all', 'rewrite'}:
            manifest['status'] = 'api_compression'
            write_json(run / 'manifest.json', manifest)
            compress_stage(config, run, tokenizer, examples, manifest, api=True)
        if args.stage in {'all', 'predict'}:
            manifest['status'] = 'predicting'
            write_json(run / 'manifest.json', manifest)
            rows = predict(config, run, tokenizer, examples, manifest)
            finish(config, run, rows, manifest)
        if args.stage in {'all', 'audit'}:
            audit(config, run, tokenizer, examples, manifest)
    except Exception as error:
        manifest.update(status='failed', error_type=type(error).__name__)
        write_json(run / 'manifest.json', manifest)
        raise


if __name__ == '__main__':
    main()

def read_compaction(run, request):
    key = fingerprint(request)
    path = run / 'aligned_compression_cache' / f'{key}.json'
    if not path.exists():
        return None
    entry = read_json(path)
    require(entry['request'] == request and entry['sha256'] == fingerprint(entry['result']), 'Compression cache corruption.')
    return {**entry['result'], 'local_compression_hash': key}

def compress_stage(config, run, tokenizer, examples, manifest, api=False):
    """先按方法遍历，GPU 一次只装一个模型；相同原文只压缩一次。"""
    contexts = list(dict.fromkeys(e['context'] for e in examples))
    if api:
        workers = min(config.get('compression_workers', 1), len(contexts))
        require(type(workers) is int and 1 <= workers <= 8, 'Use 1–8 rewrite workers.')
        stop = threading.Event()
        with ThreadPoolExecutor(max_workers=workers) as pool:
            jobs = [pool.submit(rewrite_worker, config, run, tokenizer, contexts[w::workers], manifest, w, workers, stop)
                    for w in range(workers)]
            for job in jobs:
                job.result()
        return
    for method in LOCAL_METHODS:
        engine = None
        for index, context in enumerate(contexts, 1):
            budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
            request = compression_request(context, budget, method, manifest)
            if read_compaction(run, request) is not None:
                continue
            if engine is None:
                # LLMLingua 的 seed_everything 会改环境字符串，但不会改变已启动进程的 hash seed。
                # 先核实进程从 seed=0 启动，再恢复字符串，供 SALT 的防误用检查读取。
                require(not sys.flags.hash_randomization, 'Start with PYTHONHASHSEED=0.')
                os.environ['PYTHONHASHSEED'] = '0'
                engine = create_compressor(method, config['compression_device'])
            start = time.perf_counter()
            result = align_local(engine, context, budget, tokenizer)
            save_compaction(run, request, result, time.perf_counter()-start)
            print(f'{method} [{index}/{len(contexts)}] {result["tokens"]}/{budget} tokens; clip={result["post_clip_used"]}', flush=True)
        del engine
        gc.collect()
        import torch
        torch.cuda.empty_cache()


def rewrite_worker(config, run, tokenizer, contexts, manifest, worker, workers, stop):
    """每个 worker 串行处理自己的原文分片；没有共享可写 API 账本。"""
    from blackwell.rewrite_client import PacedRewriteClient
    relative = f'compression_api/worker_{worker}'
    client = PacedRewriteClient(compression_config(config), run/relative, tokenizer, workers, worker)
    try:
        for method in API_METHODS:
            engine = LLMRewriteCompressor(client, tokenizer, method)
            for index, context in enumerate(contexts, 1):
                if stop.is_set():
                    return
                budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
                request = compression_request(context, budget, method, manifest)
                if read_compaction(run, request) is not None:
                    continue
                start = time.perf_counter()
                result = engine.compress(context, budget)
                result['compression_api_subdir'] = relative
                save_compaction(run, request, result, time.perf_counter()-start)
                print(f'{method} worker {worker} [{index}/{len(contexts)}]: {result["tokens"]}/{budget}, clip={result["post_clip_used"]}', flush=True)
    except Exception:
        stop.set()
        raise


def condition_text(context, method, tokenizer, run, manifest):
    """预测阶段只读已完成的压缩结果，不再加载压缩模型或修改压缩文本。"""
    original = tokenizer.count(context)
    budget = max(1, math.floor(.5*original))
    if method in LOCAL_METHODS or method in API_METHODS:
        result = read_compaction(run, compression_request(context, budget, method, manifest))
        require(result is not None, f'Missing compression: {method}')
        # 候选全文保存在压缩缓存，不在每道问题中重复复制。
        return {k: v for k, v in result.items() if k != 'candidates'}
    text = context if method == 'full' else ''
    if method == 'head_tail':
        text = tokenizer.head_tail(context, budget)
        return select_candidate([{'text': text, 'tokens': tokenizer.count(text)}], budget, tokenizer)
    return {'text': text, 'tokens': tokenizer.count(text), 'budget_tokens': None,
            'budget_met': None, 'length_repair_used': False, 'compression_request_hashes': []}


def predict(config, run, tokenizer, examples, manifest, offline=False):
    client = CachedChatClient(config, run, offline=offline)
    rows = []
    # 全部压缩完成后才访问 question / label；每题的所有条件配对。
    for index, example in enumerate(examples, 1):
        context = example['context']
        original = tokenizer.count(context)
        conditions = {}
        for method in config['methods']:
            c = condition_text(context, method, tokenizer, run, manifest)
            raw, key = client.complete(prediction_messages(c['text'], example['question']), config['prediction_max_tokens'], json_mode=True)
            probability = parse_probability(raw)
            conditions[method] = {**c, 'p_yes': probability, 'predicted_label': int(probability >= .5),
                                  'prediction_request_hash': key, 'retained_ratio_actual': c['tokens']/original}
        rows.append({**example, 'original_tokens': original, 'conditions': conditions})
        if not offline:
            write_json(run / 'predictions.json', rows)
            print(f'Predict [{index}/{len(examples)}] '+example['id'], flush=True)
    return rows

"""预测概率与预算的描述性诊断；不拟合校准器，不改变原始 p_yes。"""
import math

from .metrics import forecast_scores


def probability_diagnostics(probabilities, labels, bins=10):
    """等宽 reliability bins / ECE / 极端错判 / 0.5 堆积；空箱保留但不参与 ECE。"""
    forecast_scores(probabilities, labels)  # 复用严格输入校验。
    if type(bins) is not int or bins < 1:
        raise ValueError('bins must be a positive integer')
    reliability = []
    for index in range(bins):
        indices = [i for i, p in enumerate(probabilities) if min(bins-1, int(p*bins)) == index]
        count = len(indices)
        mean_p = sum(probabilities[i] for i in indices)/count if count else None
        frequency = sum(labels[i] for i in indices)/count if count else None
        reliability.append({'lower': index/bins, 'upper': (index+1)/bins, 'n': count,
                            'mean_p_yes': mean_p, 'observed_yes_rate': frequency})
    clipped_losses = {}
    for epsilon in (1e-12, 1e-6, 1e-3):
        q = [min(1-epsilon, max(epsilon, p)) for p in probabilities]
        clipped_losses[str(epsilon)] = -sum(y*math.log(p)+(1-y)*math.log1p(-p) for p, y in zip(q, labels))/len(q)
    return {'n': len(labels), 'base_rate': sum(labels)/len(labels),
            'mean_p_yes': sum(probabilities)/len(labels),
            'ece_equal_width_10': sum(b['n']/len(labels)*abs(b['mean_p_yes']-b['observed_yes_rate']) for b in reliability if b['n']),
            'reliability': reliability, 'p_half': probabilities.count(.5),
            'p_zero_or_one': sum(p in (0, 1) for p in probabilities),
            'wrong_zero_or_one': sum(p == 1-y for p, y in zip(probabilities, labels)),
            'confident_wrong_95': sum((p >= .95 and y == 0) or (p <= .05 and y == 1) for p, y in zip(probabilities, labels)),
            'distinct_probabilities': len(set(probabilities)),
            'log_loss_sensitivity': clipped_losses}

def prepare(config):
    """固定既有样本和源文件哈希；重新预测全部条件，避免混用旧日期的 forecasts。"""
    from blackwell.data import load_dataset
    from blackwell.compressors.runtime import configure_cache, runtime_identity, verify_assets
    from blackwell.tokenization import ContextTokenizer
    configure_cache()
    verify_assets(LOCAL_METHODS)
    require(config['methods'] == METHODS, 'Unexpected paired methods.')
    require(config['retained_ratio'] == .5, 'This protocol fixes 50% token cap.')
    source = ROOT / 'runs' / config['source_run']
    require(source.parent == ROOT / 'runs' and source.name != config['run_name'], 'Unsafe source path.')
    old = read_json(source / 'manifest.json')
    require(old['status'] == 'complete', 'Source run incomplete.')
    examples = read_json(source / 'examples.json')
    reloaded, provenance = load_dataset(old['identity']['config'], offline=True)
    require(examples == reloaded and fingerprint(examples) == old['identity']['examples_sha256'], 'Source samples changed.')
    require(config['sample_size'] <= len(examples), 'Cannot expand the frozen source population.')
    examples = examples[:config['sample_size']]
    snapshots = {p.name: file_hash(p) for p in source.iterdir() if p.is_file()}
    code = code_provenance()
    for path in [Path(__file__), *sorted((ROOT / 'blackwell/compressors/prompts').glob('*'))]:
        code['files_sha256'][path.relative_to(ROOT).as_posix()] = file_hash(path)
    runtime = runtime_identity(LOCAL_METHODS, config['compression_device'])
    runtime['budget_policy'] = 'length-only native-rate search <=8 calls; prefer 95-100% cap band, else closest length (feasible wins ties); explicitly head-tail clamp selected overshoots'
    identity = {'config': config, 'dataset': provenance, 'examples_sha256': fingerprint(examples),
                'code': code['files_sha256'], 'source_files_sha256': snapshots,
                'local_compressors': runtime}
    run = ROOT / 'runs' / config['run_name']
    if (run / 'manifest.json').exists():
        manifest = read_json(run / 'manifest.json')
        require(manifest['identity'] == identity, 'Code/config/data changed: choose a new run_name.')
        require(read_json(run / 'examples.json') == examples, 'Saved examples changed.')
    else:
        manifest = {'identity': identity, 'created_utc': datetime.now(timezone.utc).isoformat(), 'status': 'prepared'}
        write_json(run / 'manifest.json', manifest)
        write_json(run / 'examples.json', examples)

def compression_request(context, budget, method, manifest):
    """压缩缓存键只含上下文、预算和固定协议；绝不传整个 example。"""
    identity = manifest['identity']
    return {'context': context, 'budget': budget, 'method': method,

                                           'runtime': identity['local_compressors'],
                                           'model': identity['config']['compression_model']})}


def read_compaction(run, request):
    key = fingerprint(request)
    path = run / 'aligned_compression_cache' / f'{key}.json'
    if not path.exists():
        return None
    entry = read_json(path)
    require(entry['request'] == request and entry['sha256'] == fingerprint(entry['result']), 'Compression cache corruption.')
    return {**entry['result'], 'local_compression_hash': key}


def save_compaction(run, request, result, seconds):
    result = {**result, 'compression_seconds': seconds}
    write_json(run / 'aligned_compression_cache' / f'{fingerprint(request)}.json',
               {'request': request, 'result': result, 'sha256': fingerprint(result)})


def compress_stage(config, run, tokenizer, examples, manifest, api=False):
    """先按方法遍历，GPU 一次只装一个模型；相同原文只压缩一次。"""
    contexts = list(dict.fromkeys(e['context'] for e in examples))
    if api:
        workers = min(config.get('compression_workers', 1), len(contexts))
        require(type(workers) is int and 1 <= workers <= 8, 'Use 1–8 rewrite workers.')
        stop = threading.Event()
        with ThreadPoolExecutor(max_workers=workers) as pool:
            jobs = [pool.submit(rewrite_worker, config, run, tokenizer, contexts[w::workers], manifest, w, workers, stop)
                    for w in range(workers)]
            for job in jobs:
                job.result()
        return
    for method in LOCAL_METHODS:
        engine = None
        for index, context in enumerate(contexts, 1):
            budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
            request = compression_request(context, budget, method, manifest)
            if read_compaction(run, request) is not None:
                continue
            if engine is None:
                # LLMLingua 的 seed_everything 会改环境字符串，但不会改变已启动进程的 hash seed。
                # 先核实进程从 seed=0 启动，再恢复字符串，供 SALT 的防误用检查读取。
                require(not sys.flags.hash_randomization, 'Start with PYTHONHASHSEED=0.')
                os.environ['PYTHONHASHSEED'] = '0'
                engine = create_compressor(method, config['compression_device'])
            start = time.perf_counter()
            result = align_local(engine, context, budget, tokenizer)
            save_compaction(run, request, result, time.perf_counter()-start)
            print(f'{method} [{index}/{len(contexts)}] {result["tokens"]}/{budget} tokens; clip={result["post_clip_used"]}', flush=True)
        del engine
        gc.collect()
        import torch
        torch.cuda.empty_cache()


def rewrite_worker(config, run, tokenizer, contexts, manifest, worker, workers, stop):
    """每个 worker 串行处理自己的原文分片；没有共享可写 API 账本。"""
    from blackwell.rewrite_client import PacedRewriteClient
    relative = f'compression_api/worker_{worker}'
    client = PacedRewriteClient(compression_config(config), run/relative, tokenizer, workers, worker)
    try:
        for method in API_METHODS:
            engine = LLMRewriteCompressor(client, tokenizer, method)
            for index, context in enumerate(contexts, 1):
                if stop.is_set():
                    return
                budget = max(1, math.floor(tokenizer.count(context)*config['retained_ratio']))
                request = compression_request(context, budget, method, manifest)
                if read_compaction(run, request) is not None:
                    continue
                start = time.perf_counter()
                result = engine.compress(context, budget)
                result['compression_api_subdir'] = relative
                save_compaction(run, request, result, time.perf_counter()-start)
                print(f'{method} worker {worker} [{index}/{len(contexts)}]: {result["tokens"]}/{budget}, clip={result["post_clip_used"]}', flush=True)
    except Exception:
        stop.set()
        raise


def condition_text(context, method, tokenizer, run, manifest):
    """预测阶段只读已完成的压缩结果，不再加载压缩模型或修改压缩文本。"""
    original = tokenizer.count(context)
    budget = max(1, math.floor(.5*original))
    if method in LOCAL_METHODS or method in API_METHODS:
        result = read_compaction(run, compression_request(context, budget, method, manifest))
        require(result is not None, f'Missing compression: {method}')
        # 候选全文保存在压缩缓存，不在每道问题中重复复制。
        return {k: v for k, v in result.items() if k != 'candidates'}
    text = context if method == 'full' else ''
    if method == 'head_tail':
        text = tokenizer.head_tail(context, budget)
        return select_candidate([{'text': text, 'tokens': tokenizer.count(text)}], budget, tokenizer)
    return {'text': text, 'tokens': tokenizer.count(text), 'budget_tokens': None,
            'budget_met': None, 'length_repair_used': False, 'compression_request_hashes': []}


def predict(config, run, tokenizer, examples, manifest, offline=False):
    client = CachedChatClient(config, run, offline=offline)
    rows = []
    # 全部压缩完成后才访问 question / label；每题的所有条件配对。
    for index, example in enumerate(examples, 1):
        context = example['context']
        original = tokenizer.count(context)
        conditions = {}
        for method in config['methods']:
            c = condition_text(context, method, tokenizer, run, manifest)
            raw, key = client.complete(prediction_messages(c['text'], example['question']), config['prediction_max_tokens'], json_mode=True)
            probability = parse_probability(raw)
            conditions[method] = {**c, 'p_yes': probability, 'predicted_label': int(probability >= .5),
                                  'prediction_request_hash': key, 'retained_ratio_actual': c['tokens']/original}
        rows.append({**example, 'original_tokens': original, 'conditions': conditions})
        if not offline:
            write_json(run / 'predictions.json', rows)
            print(f'Predict [{index}/{len(examples)}] '+example['id'], flush=True)
    return rows

def audit(config, run, tokenizer, examples, manifest):
    """重新构造每条 API 请求，核验长度与 query 边界，并独立验证全部 InfoGap 单元格。"""

    rows = read_json(run / 'predictions.json')
    require(len(rows) == len(examples), 'Incomplete sample set.')
    source = ROOT / 'runs' / config['source_run']
    for name, digest in manifest['identity']['source_files_sha256'].items():
        require(file_hash(source / name) == digest, 'Old source file changed: '+name)
    checked = set()
    for row, example in zip(rows, examples):
        require({k: row[k] for k in example} == example, 'Sample pairing changed.')
        require(list(row['conditions']) == config['methods'], 'Condition mismatch.')
        for method, condition in row['conditions'].items():
            require(tokenizer.count(condition['text']) == condition['tokens'], 'Token count mismatch.')
            check_cached_prediction(run, config, row, condition)
            expected = condition_text(row['context'], method, tokenizer, run, manifest)
            require(all(condition[k] == v for k, v in expected.items()), 'Condition differs from compression cache.')
            if method in {'full', 'no_context', 'head_tail'}:
                continue
            request = compression_request(row['context'], condition['budget_tokens'], method, manifest)
            cached = read_compaction(run, request)
            key = fingerprint(request)
            if key in checked:
                continue
            checked.add(key)
            candidates = cached['candidates']
            require(all(tokenizer.count(c['text']) == c['tokens'] for c in candidates), 'Candidate length mismatch.')
            selection = select_candidate(candidates, condition['budget_tokens'], tokenizer)
            require(all(cached[k] == v for k, v in selection.items() if k != 'compression_request_hashes'), 'Length-only candidate selection mismatch.')
            if method in API_METHODS:
                previous, previous_tokens = None, None
                for candidate in candidates:
                    messages = rewrite_messages(row['context'], condition['budget_tokens'], method, previous, previous_tokens)
                    raw, request_hash = CachedChatClient(compression_config(config), run / cached['compression_api_subdir'], offline=True).complete(
                        messages, max_tokens=min(32768, max(512, 2*condition['budget_tokens']+512)))
                    require(raw == candidate['text'] and request_hash == candidate['request_hash'], 'Rewrite request/response mismatch.')
                    previous, previous_tokens = raw, candidate['tokens']
    check_matrix(rows, read_json(run / 'metrics.json'), config['methods'])
    before = {name: file_hash(run / name) for name in ('predictions.json', 'metrics.json', 'api_attempts.json')}
    with patch.object(socket.socket, 'connect', side_effect=AssertionError('Offline replay')):
        replayed = predict(config, run, tokenizer, examples, manifest, offline=True)
    require(replayed == rows, 'Offline predictions changed.')
    require(before == {name: file_hash(run / name) for name in before}, 'Replay changed files.')
    result = {'status': 'passed', 'samples': len(rows), 'forecasts': len(rows)*len(METHODS),
              'compression_contexts_checked': len(checked), 'infogap_cells': len(METHODS)**2,
              'old_files_unchanged': True, 'offline_replay_identical': True,

    print(result, flush=True)

"""统一压缩接口：公共输入只有上下文、预算和方法名。"""
from .prompts import compression_messages
from .compressors import LOCAL_METHODS


class Compactor:
    def __init__(self, client, tokenizer, local_cache=None, retained_ratio=0.5):
        self.client = client
        self.tokenizer = tokenizer
        self.local_cache = local_cache
        self.retained_ratio = retained_ratio

    def compress(self, context, budget, method):
        """故意不接收 question、label 或原文预测，确保 query-blind。"""
        if method in LOCAL_METHODS:
            if self.local_cache is None:
                raise RuntimeError("Local compressor cache/runtime was not configured.")
            local = self.local_cache.compress(context, budget, method, self.retained_ratio)
            return {**self._result(local["text"], budget, [], False), **local}
        if method in ("llm_summary", "telegrapher"):
            from .compressors.llm_rewrite import LLMRewriteCompressor
            return LLMRewriteCompressor(self.client, self.tokenizer, method).compress(context, budget)
        if method == "head_tail":
            text = self.tokenizer.head_tail(context, budget)
            return self._result(text, budget, [], False)
        if method != "summary":
            raise ValueError(f"Unknown compaction method: {method}")
        hashes = []
        previous = None
        previous_tokens = None
        for attempt in range(2):  # 最多一次长度修复；不根据预测效果挑摘要。
            messages = compression_messages(context, budget, previous, previous_tokens)
            text, request_hash = self.client.complete(
                messages, max_tokens=min(2048, max(256, budget * 3 + 100))
            )
            hashes.append(request_hash)
            count = self.tokenizer.count(text)
            if count <= budget:

DEFAULT_METHODS = ("full", "no_context", "head_tail", "summary")


def selected_methods(config):
    methods = config.get("methods", list(DEFAULT_METHODS))
    allowed = set(DEFAULT_METHODS) | set(LOCAL_METHODS) | set(API_METHODS)
    if (not isinstance(methods, list) or len(methods) != len(set(methods))
            or not {"full", "no_context"}.issubset(methods) or set(methods) - allowed):
        raise ValueError("methods must be a unique list of supported names including full and no_context.")
    return methods


def load_local_keys():
    """可选 .env 仅允许两种密钥；已有环境变量优先，不打印任何值。"""
    path = ROOT / ".env"
    if path.exists():
        for line in path.read_text(encoding="utf-8-sig").splitlines():
            name, separator, value = line.partition("=")
            if separator and name.strip() in {"SILICONFLOW_API_KEY", "OPENAI_API_KEY"}:
                os.environ.setdefault(name.strip(), value.strip().strip("\"'"))


def load_config(path):
    config = read_json(path)
    if not re.fullmatch(r"[A-Za-z0-9_-]+", config["run_name"]):
        raise ValueError("run_name must contain only letters, digits, _ or -.")
    if type(config["sample_size"]) is not int or config["sample_size"] < 1:
        raise ValueError("sample_size must be a positive integer.")
    if not 0 < config["retained_ratio"] < 1:
        raise ValueError("retained_ratio must be between 0 and 1.")
    # 第一版只验证这一组合；避免改模型后仍悄悄使用错误 tokenizer/参数。
    if (config["provider"], config["base_url"], config["model"], config["api_key_env"]) != (
        "siliconflow", "https://api.siliconflow.cn/v1", "Qwen/Qwen3-30B-A3B-Instruct-2507", "SILICONFLOW_API_KEY"
    ):
        raise ValueError("Version 0.1 supports SiliconFlow Qwen/Qwen3-30B-A3B-Instruct-2507 only. See the audit guide before adding models.")
    if config["enable_thinking"] is not False:
        raise ValueError("This pilot freezes enable_thinking=false.")
    for key in ("prediction_max_tokens", "max_api_attempts", "bootstrap_repetitions"):
        if type(config[key]) is not int or config[key] < 2:
            raise ValueError(f"Invalid setting: {key}")
    if config["min_interval_seconds"] < 0 or config["timeout_seconds"] <= 0:
        raise ValueError("Invalid timing setting.")
    selected_methods(config)
    if config.get('compression_device', 'cpu') not in {'cpu', 'cuda'}:
        raise ValueError('compression_device must be cpu or cuda.')
    return config


def code_provenance():
    paths = [ROOT / "run_experiment.py", *sorted((ROOT / "blackwell").rglob("*.py"))]
    hashes = {p.relative_to(ROOT).as_posix(): file_hash(p) for p in paths}
    result = subprocess.run(["git", "rev-parse", "HEAD"], cwd=ROOT, capture_output=True, text=True)
    return {"python": platform.python_version(), "files_sha256": hashes,
            "git_commit_at_run": result.stdout.strip() if result.returncode == 0 else None}


def prepare_run(config, run_dir, offline=False):
    # 延迟导入：离线重算报告不需要 tokenizer 或下载。
    from blackwell.tokenization import ContextTokenizer, TOKENIZER_REVISION, TOKENIZER_SHA256
    tokenizer = ContextTokenizer()
    examples, provenance = load_dataset(config, offline)
    identity = {"config": config, "dataset": provenance, "examples_sha256": fingerprint(examples),
                "tokenizer_revision": TOKENIZER_REVISION, "tokenizer_sha256": TOKENIZER_SHA256,
                "code": code_provenance()}
    local_methods = [m for m in selected_methods(config) if m in LOCAL_METHODS]
    if local_methods:
        from blackwell.compressors.runtime import configure_cache, runtime_identity, verify_assets
        configure_cache()
        verify_assets(local_methods)
        identity["local_compressors"] = runtime_identity(local_methods, config.get('compression_device', 'cpu'))
    manifest_path = run_dir / "manifest.json"
    if manifest_path.exists():
        manifest = read_json(manifest_path)
        # git 提交与否不改变语义；代码内容和配置改变时必须换 run_name。
        old_identity = manifest["identity"]
        comparable_old = {**old_identity, "code": old_identity["code"]["files_sha256"]}
        comparable_new = {**identity, "code": identity["code"]["files_sha256"]}
        if comparable_old != comparable_new:
            raise ValueError("This run_name already belongs to different code/config/data. Choose a new run_name.")
    else:
        manifest = {"identity": identity, "created_utc": datetime.now(timezone.utc).isoformat(), "status": "prepared"}
        write_json(manifest_path, manifest)
        write_json(run_dir / "examples.json", examples)
    return tokenizer, examples, manifest


def generate_predictions(config, run_dir, tokenizer, examples, offline=False):
    client = CachedChatClient(config, run_dir, offline)
    methods = selected_methods(config)
    local_cache = None
    if any(m in LOCAL_METHODS for m in methods):
        from blackwell.local_cache import LocalCompressionCache
        identity = read_json(run_dir / "manifest.json")["identity"]["local_compressors"]
        local_cache = LocalCompressionCache(run_dir, identity, offline)
    compactor = Compactor(client, tokenizer, local_cache, config["retained_ratio"])
    rows = []
    for number, example in enumerate(examples, 1):
        context = example["context"]
        original_tokens = tokenizer.count(context)
        budget = max(1, math.floor(config["retained_ratio"] * original_tokens))
        # 所有压缩先完成，再让 reader 看到问题。compress() 不接收整个 example。
        compacted = {
            "full": {"text": context, "tokens": original_tokens, "budget_tokens": None,
                     "budget_met": None, "length_repair_used": False, "compression_request_hashes": []},
            "no_context": {"text": "", "tokens": 0, "budget_tokens": None,
                           "budget_met": None, "length_repair_used": False, "compression_request_hashes": []},
        }
        for method in methods:
            if method in {"full", "no_context"}:
                continue
            compacted[method] = compactor.compress(context, budget, method)
        conditions = {}
        for method in methods:
            compressed = compacted[method]
            raw, request_hash = client.complete(
                prediction_messages(compressed["text"], example["question"]),
                max_tokens=config["prediction_max_tokens"], json_mode=True,
            )
            probability = parse_probability(raw)
            conditions[method] = {
                **compressed, "p_yes": probability, "predicted_label": int(probability >= 0.5),
                "prediction_request_hash": request_hash,
                "retained_ratio_actual": compressed["tokens"] / original_tokens,
            }
        rows.append({**example, "original_tokens": original_tokens, "conditions": conditions})
        write_json(run_dir / "predictions.json", rows)
        print(f"[{number:02d}/{len(examples):02d}] {example['id']}  " +
              "  ".join(f"{m}={conditions[m]['p_yes']:.3f}" for m in methods), flush=True)
    return rows, {"new_api_attempts": client.new_attempts, "cache_hits": client.cache_hits}


def analyze(rows, config):
    if len(rows) != config["sample_size"] or len({r["id"] for r in rows}) != len(rows):
        raise ValueError("Incomplete or duplicate samples: refusing to report a silently reduced dataset.")
    methods = selected_methods(config)
    if any(set(r["conditions"]) != set(methods) for r in rows):
        raise ValueError("Conditions must match the configured methods on every sample.")
    labels = [r["label"] for r in rows]
    groups = [r["context_group"] for r in rows]
    probabilities = {method: [r["conditions"][method]["p_yes"] for r in rows] for method in methods}
    scores = {}
    for method in methods:
        conditions = [r["conditions"][method] for r in rows]
        # 论文的多个问题复用同一压缩结果；耗时不能按题目重复累计。
        unique_compressions = {c['local_compression_hash']: c['compression_seconds']
                               for c in conditions if 'local_compression_hash' in c}
        scores[method] = {
            **forecast_scores(probabilities[method], labels),
            "mean_context_tokens": sum(c["tokens"] for c in conditions) / len(rows),
            "pooled_retained_ratio": sum(c["tokens"] for c in conditions) / sum(r["original_tokens"] for r in rows),
            "over_budget_samples": sum(c["budget_met"] is False for c in conditions),
            "length_repairs": sum(c["length_repair_used"] for c in conditions),
            "unique_compressions": len(unique_compressions),
            "compression_seconds_unique_total": sum(unique_compressions.values()),
        }
    for method in methods:
        for metric in ('accuracy', 'brier', 'log_loss'):
            scores[method][metric + '_delta_vs_full'] = scores[method][metric] - scores['full'][metric]
    matrix = {a: {b: infogap(probabilities[a], probabilities[b], labels) for b in methods} for a in methods}
    uncertainty = {method: bootstrap_gaps(
        probabilities["full"], probabilities[method], labels, groups,
        config["bootstrap_repetitions"], config["bootstrap_seed"],
    ) for method in methods if method != "full"}
    curves = {m: payoff_curve(probabilities['full'], probabilities[m], labels)
              for m in methods if m != 'full'}
    subsets = {}
    for task in sorted({r['source_task'] for r in rows if 'source_task' in r}):
        indices = [i for i, r in enumerate(rows) if r.get('source_task') == task]
        y = [labels[i] for i in indices]
        p = {m: [probabilities[m][i] for i in indices] for m in methods}
        subsets[task] = {'n': len(indices), 'positive_labels': sum(y),
                         'scores': {m: forecast_scores(p[m], y) for m in methods},
                         'infogap_vs_full': {m: {'full_over_method': infogap(p['full'], p[m], y),
                                                'method_over_full': infogap(p[m], p['full'], y)}
                                             for m in methods if m != 'full'}}
    return {"n": len(rows), "positive_labels": sum(labels), "scores": scores,
            "context_clusters": len(set(groups)), "payoff_curves_vs_full": curves, "source_tasks": subsets,
            "infogap": matrix, "bootstrap_vs_full": uncertainty,
            "normalization": "2 * sup_t mean[(t-y)(1[p_A<=t]-1[p_B<=t])]; also save unscaled.",
            "scope": "Small pilot of verbalized probabilities, not true-posterior recovery or a dominance test."}


def write_outputs(run_dir, rows, config):
    from blackwell.reporting import write_report
    methods = selected_methods(config)
    metrics = analyze(rows, config)
    write_json(run_dir / "metrics.json", metrics)
    fields = ["id", "label", "method", "p_yes", "p_no", "predicted_label", "context_tokens", "budget_tokens", "retained_ratio", "budget_met"]
    with (run_dir / "predictions.csv").open("w", encoding="utf-8-sig", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        for row in rows:
            for method in methods:
                c = row["conditions"][method]
                writer.writerow({"id": row["id"], "label": row["label"], "method": method,
                                 "p_yes": c["p_yes"], "p_no": 1 - c["p_yes"], "predicted_label": c["predicted_label"],
                                 "context_tokens": c["tokens"], "budget_tokens": c["budget_tokens"],
                                 "retained_ratio": c["retained_ratio_actual"], "budget_met": c["budget_met"]})

    return metrics


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--config", default="configs/pilot.json")
    modes = parser.add_mutually_exclusive_group()
    modes.add_argument("--prepare-only", action="store_true", help="Download dataset/tokenizer, but do not call the LLM.")
    modes.add_argument("--offline", action="store_true", help="Replay exactly the saved API responses; fail on cache misses.")
    modes.add_argument("--report-only", action="store_true", help="Recompute metrics/report from saved predictions, without network.")
    args = parser.parse_args()
    config_path = Path(args.config)
    if not config_path.is_absolute():
        config_path = ROOT / config_path
    config = load_config(config_path)
    run_dir = ROOT / "runs" / config["run_name"]
    if args.report_only:
        saved_config = read_json(run_dir / "manifest.json")["identity"]["config"]
        if saved_config != config:
            raise ValueError("Report config differs from the saved experiment config.")
        write_outputs(run_dir, read_json(run_dir / "predictions.json"), config)
        print(f"Report: {run_dir / 'report.html'}")
        return
    if args.offline:
        # 离线重放必须已准备好数据，不允许触发 download() 的网络分支。
        required = [ROOT / "data" / "tokenizer" / "qwen3-30b-a3b-instruct-2507.json",

            raise RuntimeError("Offline replay needs the previously prepared local files.")
    tokenizer, examples, manifest = prepare_run(config, run_dir, args.offline)
    print(f"{len(examples)} {manifest['identity']['dataset']['name']} {config['dataset_split']} samples; {len(selected_methods(config))} paired conditions.")
    if args.prepare_only:
